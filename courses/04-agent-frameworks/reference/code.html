"""真实 LangGraph Send 与 reducer；工作函数固定，离线检验编排。"""

# 一个文档对应一个工作者输入；本例验证真实图的动态分发，不调用生成模型。
import argparse
import json
import operator
# Annotated 在类型上附带额外处理规则；这里的 reducer 决定状态更新时是追加消息还是覆盖旧消息。
# TypedDict 描述共享状态有哪些字段，便于读代码和静态检查；它本身不会替你做运行时数据验证。
from typing import Annotated, TypedDict
from langgraph.graph import StateGraph, START, END
from langgraph.types import Send


# checks 使用加法 reducer 合并并行结果，避免后返回的结果覆盖前一个。
class State(TypedDict):
    documents: list[dict]
    checks: Annotated[list[dict], operator.add]
    report: dict


# 每个工作者只接收自己负责的文档，避免依赖未分配的共享输入。
class WorkerState(TypedDict):
    document: dict


# 先拒绝空输入和重复 ID，否则汇总时无法判断有没有漏查或覆盖。
def validate(state):
    documents = state['documents']
    ids = [doc['id'] for doc in documents]
    if not documents or len(ids) != len(set(ids)):
        raise ValueError('文档不能为空，ID 不能重复')
    return {'report': {}}


# 文档数量决定 Send 的数量；每条 Send 指定节点名和该节点收到的状态。
def dispatch(state):
    return [Send('check_document', {'document': doc}) for doc in state['documents']]


# 只检查约定的两个词，返回状态更新；不要把它当内容正确性审查。
def check_document(state: WorkerState):
    doc = state['document']
    missing = [word for word in ('目的', '运行') if word not in doc['text']]
    return {'checks': [{'id': doc['id'], 'missing': missing, 'ok': not missing}]}


# 先核对结果数量，再按 ID 排序；只有全部工作者通过，整份报告才通过。
def merge(state):
    checks = sorted(state['checks'], key=lambda row: row['id'])
    if len(checks) != len(state['documents']):
        raise ValueError('工作者结果不完整')
    return {'report': {'status': 'PASS' if all(x['ok'] for x in checks) else 'FAIL',
                       'checks': checks}}


def run(documents):
    # 以 State 作为图中各节点共享的数据约定；每个节点返回自己的更新，数据通过状态传递而非隐式全局变量。
    graph = StateGraph(State)
    # 把 validate 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    graph.add_node('validate', validate)
    # 把 check_document 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    graph.add_node('check_document', check_document)
    # 把 merge 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    graph.add_node('merge', merge)
    # START 完成后进入 validate，将这两步的执行先后固定在程序中。
    graph.add_edge(START, 'validate')
    # dispatch 返回的 Send 决定这次有多少工作者，不是提前固定两条分支。
    # 由路由函数返回目标名称，再通过这张映射选边；返回了未登记的名称就不能继续，路径必须可核对。
    graph.add_conditional_edges('validate', dispatch, ['check_document'])
    # check_document 完成后进入 merge，将这两步的执行先后固定在程序中。
    graph.add_edge('check_document', 'merge')
    # merge 完成后进入 END，将这两步的执行先后固定在程序中。
    graph.add_edge('merge', END)
    # 把节点与连线变成可运行的图；这一步是组装流程，还没有替用户执行一次任务。
    return graph.compile().invoke({'documents': documents, 'checks': [], 'report': {}})['report']


if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    # 为运行者提供明确的选项与默认值；这里的限制让同一实验可以按相同条件重跑。
    parser.add_argument('--fixed', action='store_true')
    # 先把命令行输入转为受约束的参数，防止错误模式或缺少参数到执行一半才暴露。
    args = parser.parse_args()
    docs = [{'id': 'a', 'text': '目的：演示。运行：python app.py'},
            {'id': 'b', 'text': '目的：核查。' + ('运行：python check.py' if args.fixed else '')}]
    # 以 JSON 保留字段和类型，便于重跑比较或人工核对；不能只保存对象的调试显示。
    # ensure_ascii：设为 False 时直接保留中文，方便阅读日志与工具结果；JSON 结构和字段含义不变。
    # indent：给本地 JSON 产物增加缩进，方便人工逐字段检查；不会改变数据内容。
    print(json.dumps(run(docs), ensure_ascii=False, indent=2))

"""五种编排的可运行控制流样例；用固定函数隔离模型随机性。"""

# 五个函数分别演示五种控制流程；输入和检查均固定，便于先看懂步骤如何连接。
# 线程池让独立的等待型调用重叠执行；它不能让有前后数据依赖的工具提前运行。
from concurrent.futures import ThreadPoolExecutor
import json


# 前一步没有产物就停止，避免把空输入继续传给后面的步骤。
def chain(text):
    draft=text.strip()
    if not draft:
        return {'status':'blocked','reason':'第一阶段没有产物'}
    return {'status':'done','summary':draft[:30]}


# 只处理已登记的任务类别；未知类别返回澄清信息。
def route(task):
    handlers={'事实核对':lambda:{'expert':'证据检查','result':'必须附来源'},
              '格式核对':lambda:{'expert':'格式检查','result':'必须有标题'}}
    if task not in handlers:
        return {'expert':None,'result':'未支持的任务，请澄清'}
    return handlers[task]()


# 两项检查彼此独立，分别返回不同字段，再合并结果。
def parallel(text):
    checks=[lambda:{'format':text.startswith('# ')},lambda:{'source':'[d1]' in text}]
    with ThreadPoolExecutor(max_workers=2) as pool:
        results=list(pool.map(lambda fn:fn(),checks))
    return {key:value for result in results for key,value in result.items()}


# 先限制任务数并去重，再给每项任务分配来源标识。
def orchestrate(topics):
    if len(topics)>3 or len(topics)!=len(set(topics)):
        raise ValueError('任务最多三个且不能重复')
    jobs=[{'topic':topic,'source':'d'+str(i+1)} for i,topic in enumerate(topics)]
    return [{'topic':job['topic'],'draft':'需核对 '+job['topic'],'source':job['source']} for job in jobs]


# 每轮记录具体缺项，再补相应内容；轮数耗尽时返回 stopped。
def optimize(text, limit=2):
    history=[]
    for attempt in range(limit+1):
        missing=[]
        if not text.startswith('# '): missing.append('标题')
        if '[d1]' not in text: missing.append('引用')
        history.append({'attempt':attempt,'missing':missing})
        if not missing: return {'status':'done','text':text,'history':history}
        if attempt==limit: break
        if '标题' in missing: text='# 摘要\n'+text
        if '引用' in missing: text+=' [d1]'
    return {'status':'stopped','text':text,'history':history}

# 节点只返回自己更新的字段，状态会逐步补齐，因此 total=False 允许字段暂时缺失；TypedDict 不负责运行时拦截坏数据。
class State(TypedDict, total=False):
    topic: str
    # draft：保存尚未批准的候选内容，把“生成成功”和“允许发布”分成两个不同阶段。
    draft: str
    source_ids: list[str]
    checked: bool
    status: str
    published: bool


def summarize(state):
    # 固定摘录让本题只考流程；真实生成可以替换此节点，仍须走后续检查。
    return {'draft': SOURCES.get(state['topic'], ''),
            'source_ids': [state['topic']] if state['topic'] in SOURCES else [], 'published': False}


# 本题要求草稿等于固定资料摘录，来源存在且内容匹配才进入人工审核。
def check_sources(state):
    refs = state['source_ids']
    valid = bool(refs) and all(x in SOURCES for x in refs)
    valid = valid and state['draft'] == '\n'.join(SOURCES[x] for x in refs)
    return {'checked': valid, 'status': 'awaiting_review' if valid else 'blocked'}


# interrupt 暂停后等待外部传入布尔决定；恢复时本节点可能从头再跑。
def review(state):
    # 把待审草稿带到暂停事件中，等待人给出明确决定；恢复时该节点可能重新执行，暂停前不要放不可重复的外部动作。
    decision = interrupt({'draft': state['draft'], 'source_ids': state['source_ids'],
                          'action': '仅标记本地接受，不向外发布'})
    if type(decision) is not bool:
        raise ValueError('批准必须为布尔值')
    return {'status': 'accepted' if decision else 'rejected', 'published': decision}


# 来源检查失败直接结束，通过才进入 review；检查点由调用者传入。
def build(saver):
    # 以 State 作为图中各节点共享的数据约定；每个节点返回自己的更新，数据通过状态传递而非隐式全局变量。
    g = StateGraph(State)
    # 把 summarize 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    g.add_node('summarize', summarize)
    # 把 check 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    g.add_node('check', check_sources)
    # 把 review 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    g.add_node('review', review)
    # START 完成后进入 summarize，将这两步的执行先后固定在程序中。
    g.add_edge(START, 'summarize')
    # summarize 完成后进入 check，将这两步的执行先后固定在程序中。
    g.add_edge('summarize', 'check')
    # 由路由函数返回目标名称，再通过这张映射选边；返回了未登记的名称就不能继续，路径必须可核对。
    g.add_conditional_edges('check', lambda s: 'review' if s['checked'] else END)
    # review 完成后进入 END，将这两步的执行先后固定在程序中。
    g.add_edge('review', END)
    # 把节点与连线变成可运行的图；这一步是组装流程，还没有替用户执行一次任务。
    # checkpointer：把图的中间状态交给检查点保存器；暂停后恢复需要这份状态，单靠再次调用图无法找回进度。
    return g.compile(checkpointer=saver)


# 同一数据库和 thread_id 才指向同一个任务；已结束任务不能再次批准。
def execute(database, thread, topic='course-3', decision=None):
    # 运行配置必须在暂停和恢复时保持同一个会话标识，否则恢复会找不到刚才待审核的状态。
    # configurable：传递可在每次运行时指定的图配置；会话标识放在这里，供检查点系统定位历史状态。
    # thread_id：定位同一条工作流的检查点。暂停和恢复必须使用同一个值；换值通常会进入另一条会话。
    config = {'configurable': {'thread_id': thread}}
    # 检查点写入磁盘，下一进程使用同一数据库和 thread_id 才能继续。
    # 打开检查点数据库连接；路径决定保存在哪里，后续进程必须打开同一个库才能恢复。
    with SqliteSaver.from_conn_string(str(database)) as saver:
        graph = build(saver)
        # 读取指定会话已经保存的状态，用来确认暂停位置和待审核内容，而不是重新生成草稿。
        snapshot = graph.get_state(config)
        if decision is None:
            if snapshot.values:
                raise ValueError('任务编号已存在；请恢复任务或换新编号')
            graph.invoke({'topic': topic}, config)
        else:
            if not snapshot.next or snapshot.values.get('status') != 'awaiting_review':
                raise ValueError('该任务不在等待批准状态，不能重复或越过批准')
            # 把外部决定交回原暂停点，不重新创建一个跳过检查的新任务。
            graph.invoke(Command(resume=decision), config)
        # 读取指定会话已经保存的状态，用来确认暂停位置和待审核内容，而不是重新生成草稿。
        snapshot = graph.get_state(config)
        return {'state': snapshot.values, 'next': list(snapshot.next)}

## 两个 framework 的并排比较

| 维度 | LangGraph | CrewAI |
|---|---|---|
| 核心抽象 | `StateGraph` + node + edge | `Agent` + `Task` + `Crew` |
| 思考方式 | “状态怎么流动” | “角色怎么分工” |
| Loop 控制 | 显式 conditional edge | 隐藏在 `Crew.kickoff()` 里 |
| Debug 路径 | 看 graph state 与 checkpoint | 看 task output 与 verbose log |
| 适合场景 | 需要明确状态与分支的 workflow | 用角色与任务快速表达合作流程 |
| 学习曲线 | 中-高 | 低 |

### LangGraph 风格（精简）

```python
g = StateGraph(State)
g.add_node("agent", agent_node)
g.add_node("tools", tool_node)
g.add_conditional_edges("agent", should_continue, {"tools": "tools", END: END})
g.add_edge("tools", "agent")
```

“我要显式地告诉系统：状态长这样、节点互相连这样、条件分支看 `should_continue`。”

### CrewAI 风格（精简）

```python
researcher = Agent(role="Researcher", goal="...", tools=[search], llm=MODEL)
task = Task(description=query, expected_output="...", agent=researcher)
crew = Crew(agents=[researcher], tasks=[task])
crew.kickoff()
```

“我要描述：这个角色是谁、要完成什么任务、有什么工具。框架自己决定怎么跑。”

## 观察重点

1. **抽象代价**：CrewAI 隐藏的多、写得少；要 debug 时 stack 比较深
2. **小 model 友善度**：两边都要实测；角色描述、工具 schema 与任务长度都会影响结果
3. **可控性**：LangGraph 你能看到每个 state 变化；CrewAI 偏向“结果导向”
4. **何时选哪个**：需要逐步看状态时先试 LangGraph；想先表达角色分工时先试 CrewAI，再用自己的任务测量

## 常见坑

- **LangGraph `bind_tools`**：要 `llm.bind_tools([search])` 才会把 tool schema 给 LLM。没 bind 模型就不知道 tool 存在
- **CrewAI LLM 设定**：要用 LiteLLM 格式（譬如 `"ollama/qwen2.5:3b"`、不是 `"qwen2.5:3b"`）。写错 provider 前缀可能连到不同后端，所以执行前要打印设定并确认
- **CrewAI 结果型别**：`crew.kickoff()` 回 `CrewOutput` 对象、`str(result)` 拿文字。直接 `print(result)` 有可能拿到 repr

from langchain_core.tools import tool

from langgraph.graph.message import add_messages

# TypedDict 描述共享状态有哪些字段，便于读代码和静态检查；它本身不会替你做运行时数据验证。
from typing_extensions import Annotated, TypedDict

MODEL = os.environ.get("MODEL", "gemma4:e4b")


# === 模型可请求的工具 ===

@tool
def search(query: str) -> str:
    """按主题检索固定的离线教学资料；支持“台北”和 react agent 两个查询键。"""
    db = {
        "台北": "台北是台湾地区的城市，示例人口约 260 万，以夜市闻名。",
        "react agent": "ReAct（推理与行动）是一种让判断、行动与观察交替进行的 Agent 模式。",
    }
    return db.get(query.strip().lower(), f"没有此条目：{query}")


# === LangGraph 状态定义 ===

class State(TypedDict):
    # 把 add_messages 作为合并规则：新一轮消息追加到历史，而不是替换整段对话，工具结果才不会丢失。
    messages: Annotated[list, add_messages]


# 把节点和条件边连接成图；阅读分支函数，确认什么情况下继续、暂停或结束。
def build_graph(llm: Any) -> Any:
    """组一个 search → summarize 的图：LLM 看 query 决定要不要调用 search、看到 tool result 后给摘要。"""
    # 把工具说明提供给模型，使它能提出结构化调用；真正执行仍由后面的工具节点负责。
    llm_with_tools = llm.bind_tools([search])

    def agent_node(state: State):
        return {"messages": [llm_with_tools.invoke(state["messages"])]}

    def tool_node(state: State):
        msg = state["messages"][-1]
        results = []
        for call in msg.tool_calls:
            name = call.get("name")
            args = call.get("args")
            # 模型给出的名称只是请求；执行前再次校验白名单，防止模型请求一个未授权函数。
            if name != "search":
                raise ValueError(f"未知工具：{name!r}")
            # 把输入边界一次检查清楚：必须是对象、只能有 query、值必须是字符串。Schema 描述本身不能替执行器做这项检查。
            if not isinstance(args, dict) or set(args) != {"query"} or not isinstance(args["query"], str):
                raise ValueError("search 只接受一个名为 query 的字符串参数。")
            # 直到名称和参数都通过验证才真正调用工具；返回值是观察数据，不是新的控制指令。
            obs = search.invoke(args)
            # 使用原请求 ID 封装工具观察，模型才能知道这份结果回答哪一次调用；多工具时尤其不能省略关联。
            results.append(ToolMessage(content=obs, tool_call_id=call["id"]))
        return {"messages": results}

    def should_continue(state: State) -> str:
        # 最新模型消息仍含工具请求就进入执行节点，否则结束；退出依据是协议字段，不是模型文字中写了“完成”。
        return "tools" if state["messages"][-1].tool_calls else END

    # 以 State 作为图中各节点共享的数据约定；每个节点返回自己的更新，数据通过状态传递而非隐式全局变量。
    g = StateGraph(State)
    # 把 agent 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    g.add_node("agent", agent_node)
    # 把 tools 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    g.add_node("tools", tool_node)
    # 把这个节点设为新任务的入口，明确第一步由谁处理。
    g.set_entry_point("agent")
    # 由路由函数返回目标名称，再通过这张映射选边；返回了未登记的名称就不能继续，路径必须可核对。
    g.add_conditional_edges("agent", should_continue, {"tools": "tools", END: END})
    # 执行工具后回到模型，让模型依据观察决定继续调用还是给出答案。
    g.add_edge("tools", "agent")
    # 流程图到这里才完成组装；实际请求要等 run 中的 invoke，便于先检查图结构再执行。
    return g.compile()


def run(query: str, llm: Any = None) -> dict:
    """Public entrypoint: 跑一轮 agent、返回 final message + steps。"""

def search_knowledge_base(query: str) -> str:

def search(query: str) -> str:

    """CrewAI 风格：一个 agent + 一个 task。"""
    os.environ["OPENAI_API_BASE"] = f"{OLLAMA_BASE}/v1"
    os.environ["OPENAI_API_KEY"] = "ollama"

    researcher = Agent(
        # role：说明这名 Agent 的职责，帮助拆分工作；角色名称本身不会自动创建独立权限或可靠协作。
        role='研究者',
        # goal：告诉这个角色要完成什么成果；与角色名称相比，具体目标更能决定它会寻找哪些信息。
        goal="查找指定主题，并用中文概括检索到的资料。",
        # backstory：给模型补充角色职责与工作方式；它影响生成行为，但不能替代执行层的工具权限检查。
        backstory="你负责检索资料库，并用中文给出简洁摘要。",
        # tools：只向模型公开本轮需要的工具说明。模型提出调用后，执行层仍须检查名称、参数与权限。
        tools=[search],
        llm=llm_model,
        # verbose：关闭框架的详细过程输出，减少噪声；需要调试时查看明确记录的轨迹，而不是把终端文字当完整审计。
        verbose=False,
    )

        description=query,
        # expected_output：说明任务应交付什么，方便下一角色消费和人工核对；这段文字并不是自动验证器。
        expected_output="依据检索结果写成的一到两句中文摘要。",
        # agent：明确由哪个角色执行任务，防止任务说明存在却没有实际执行者。
        agent=researcher,
    )

    return Crew(agents=[researcher], tasks=[task], verbose=False)

### Agent

### Task

```python
research_task = Task(
    description="Search for X and report findings.",
    expected_output="A 1-2 sentence factual entry.",
    agent=researcher,
)
```

**重点**：`expected_output` 是给 LLM 看的“合格范本”。写成“两句、主动语态的开场”比“一些文字”更清楚；改善幅度要用自己的任务评测。

### Context dependency

```python
write_task = Task(..., context=[research_task]) # writer 看 researcher 结果
critic_task = Task(..., context=[research_task, write_task]) # critic 同时看两个
```

**重点**：`context` 是 CrewAI 的 dataflow 机制。`critic_task.context=[a, b]` 表示 critic 看到 a, b 两个 task 的 output。

### Sequential vs Hierarchical Process

```python
Crew(..., process=Process.sequential) # 线性走完
Crew(..., process=Process.hierarchical) # 多个 manager+worker、需设 manager_llm
```

这题用 sequential，因为顺序最容易看懂。Hierarchical 会由 manager 分派任务，适合需要动态分工、而且已有评测与停止条件的场景。

## 两个 path 观察重点

| 观察项 | Anthropic Claude haiku | Ollama qwen2.5:3b |
|---|---|---|
| Researcher 是否调用 tool | 看 log 与输出验证 | 看 log 与输出验证 |
| Writer 是否使用研究结果 | 用测试案例检查 | 用相同测试案例检查 |
| Critic 是否抓到错误 | 不预设一定成功 | 不预设一定成功 |
| 速度 | 在自己的网络与任务量测 | 在自己的硬体与模型量测 |
| 模型 API 成本 | 依 tokens 与调用次数计算 | $0 |

**教学 punchline**：multi-agent 多了交接点；任何角色漏掉资讯，错误都可能往后传。模型大小不是唯一答案，还要测角色 prompt、工具结果、handoff 与停止条件。

## 常见坑

- **`expected_output` 太笼统**：写“Some output”没有清楚成功条件。改成“A 2-sentence blog intro paragraph in active voice”，再用测试案例比较
- **`context` 漏设**：Writer 没设 `context=[research_task]`、就拿不到 researcher 结果、会凭空写
- **小 model + 3 agent**：可能比较慢或漏步。先看 log；需要时再比较 `qwen2.5:7b` 或 Claude
- **`allow_delegation=True` 慎用**：开启后 agent 可以叫其他 agent 帮忙、容易 loop。雏形阶段建议 `False`

## 延伸

- **加 manager**：`process=Process.hierarchical` + `manager_llm=...`、让 manager agent 动态分配
- **加 memory**：CrewAI 有 `memory=True`、让 agent 跨 task 记住 context
- **批次或异步执行**：`crew.kickoff_for_each(...)` 处理一批输入，`crew.kickoff_async(...)` 异步执行；两者都不是 streaming
- **加 streaming**：创建 `Crew(..., stream=True)`，再调用 `crew.kickoff()`
- **加 human-in-the-loop**：本题用练习 3 的 LangGraph 示范；CrewAI 也有自己的 human-feedback triggers

## 任务

`classify → [search?] → respond → [HITL] → final`

- **`classify_node`**：看 query 决定 `needs_search`
- **条件分支**：`needs_search=True` 走 `search` node、否则直接 `respond`
- **HITL checkpoint**：`review_node` 用 `interrupt()` 暂停，等待人类回答
- **`final_node`**：`approved=True` → PUBLISHED、否则 REJECTED

这题用 LangGraph 示范 **graph state（图状态）**、**checkpoint（检查点）**、`interrupt()` 与 `Command(resume=...)`。你可以看见流程停在哪里，以及它如何从同一个 `thread_id` 继续。

## LangGraph 图结构（精简）

```python
g = StateGraph(State)
g.add_node("classify", classify_node)
g.add_node("search", search_node)
g.add_node("respond", respond_node)
g.add_node("final", final_node)

g.add_edge(START, "classify")
g.add_conditional_edges("classify", should_search, {"search": "search", "respond": "respond"})
g.add_edge("search", "respond")
g.add_edge("respond", "review")
g.add_edge("review", "final")
g.add_edge("final", END)

graph = g.compile(checkpointer=InMemorySaver())
```

## HITL 怎么运作

```python
# 第一段：review_node 调用 interrupt()，图就把资料存进 checkpoint 并停下来
config = {"configurable": {"thread_id": "demo"}}
state_before = graph.invoke({"query": ...}, config=config)
# state_before["__interrupt__"] 会带出草稿与问题

# 第二段：人类回答后，用同一个 thread_id 恢复
state_after = graph.invoke(Command(resume=True), config=config)
```

**关键**：`interrupt()` 是“先停一下”；`Command(resume=True/False)` 是“带着人的回答继续”。Production 可以把这个等待点接到 webhook、Slack 或前端按钮。

## 为什么这个 pattern 重要

| 情境 | 不用 HITL | 用 HITL |
|---|---|---|
| Agent 发 email | 直接送出（风险） | 显示草稿、人类按 approve |
| Agent 改 production 设定 | 直接套用 | dry-run 后等核准 |
| Agent 做退款 | 自动退 | 超过 $X 等审核 |

有 **side effect（会改变外部世界的动作）** 时，先判断风险；寄信、退款或改 production 设定通常需要 HITL、权限检查与 audit log。低风险唯读动作不一定要每次人工核准。

## 两个 path 观察重点

两个 path 共用同一张图；`classify`、离线查询与路由是可预测的 Python，`respond` 则真的调用不同模型写草稿。**比较时只换模型，不要同时改 graph。**

Node 里只负责暂停，拿到人的答案后再更新 state：

```python
from langgraph.types import interrupt

def review_node(state):
    approved = interrupt({"draft": state["draft"], "question": "Approve?"})
    return {"approved": approved}
```

外面的 caller 收到真人答案后，才从同一个 `thread_id` 继续：

```python
from langgraph.types import Command

human_answer = True
result = graph.invoke(Command(resume=human_answer), config=config)
```

## 常见坑

- **`checkpointer` 没设**：没有 checkpointer 就无法可靠地保存 pause/resume 状态
- **`thread_id` 不一致**：第一段 `invoke` 与 `Command(resume=...)` 必须用同一个设定，否则找不到原来的 checkpoint
- **在 `interrupt()` 前做 side effect**：恢复时节点可能重新执行。把寄信或扣款放在核准后，并加入 idempotency key
- **conditional_edges 函数要回 string**：`should_search` return value 必须是 `add_conditional_edges` 第三个参数 dict 的 key、不能 return literal value 直接当 node name

## 想看更聪明的答案？

比较另一个模型，或把记忆体 checkpointer 换成适合部署环境的持久化储存。先确认官方 persistence 文件与失败恢复行为，再选资料库。

## 延伸

- **加 retry**：在 `search_node` 失败时 retry、用 LangGraph 的 `error` edge
- **加多个 HITL**：在不同 review node 调用 `interrupt()`，并替每个核准动作定义清楚资料
- **time-travel debug**：`graph.get_state_history(config)` 拿到所有 checkpoint、可以回到任一步 fork 新 thread
- **加 streaming**：`for state in graph.stream(...)` 边跑边看 state

"""Stage 4 练习 3：用 LangGraph 做条件分支与真正可暂停的 HITL。

流程像一条有岔路的轨道：先判断要不要查资料，再请模型写草稿，
接着用 ``interrupt()`` 停下来等人决定。只有收到 ``Command(resume=...)``
之后，图才会继续走到发布或拒绝。Path A 使用 Ollama；Path B 共用同一张图。
"""

# 先分类选择是否检索，再暂停等待决定；内存检查点只保留在当前进程中。

from __future__ import annotations

import os
import sys
# Literal 把允许返回的路由名写进类型说明，帮助核对条件边是否有对应目标。
from typing import Any, Literal

# 不同终端对标准输出对象的实现不同，先确认支持重设编码，再开启中文输出，避免测试替身上报属性错误。
if hasattr(sys.stdout, "reconfigure"):
    # 统一终端的中文编码；无法编码的字符只替换显示，不能让日志打印失败掩盖实际业务结果。
    sys.stdout.reconfigure(encoding="utf-8", errors="replace")

from langchain_openai import ChatOpenAI
# 内存检查点能在同一进程内暂停和恢复；进程退出就丢失，跨进程演示应使用后面的磁盘保存器。
from langgraph.checkpoint.memory import InMemorySaver
from langgraph.graph import END, START, StateGraph
# Command 携带恢复值或跳转信息；恢复暂停节点时要将人的决定放入 resume。
# interrupt 会把图暂停在审核点并保存待处理信息；收到恢复值之前，后续节点不会继续执行。
from langgraph.types import Command, interrupt
# TypedDict 描述共享状态有哪些字段，便于读代码和静态检查；它本身不会替你做运行时数据验证。
from typing_extensions import TypedDict

MODEL = os.environ.get("MODEL", "gemma4:e4b")


# 节点只返回自己更新的字段，状态会逐步补齐，因此 total=False 允许字段暂时缺失；TypedDict 不负责运行时拦截坏数据。
class State(TypedDict, total=False):
    # query：保留最初的问题，后续分类、检索与生成都围绕它工作；不要把检索结果覆盖成新的用户问题。
    query: str
    # needs_search：分类节点写入的路由决定，条件边据此选择检索或直接回答；这是程序规则，不是模型自行授权。
    needs_search: bool
    # search_result：保存实际查到的资料，生成节点必须读取这个字段，才有依据给出草稿。
    search_result: str
    # draft：保存尚未批准的候选内容，把“生成成功”和“允许发布”分成两个不同阶段。
    draft: str
    # approved：保存人工审核的布尔决定；必须明确为真或假，不能把任意非空字符串当作同意。
    approved: bool
    # final：保存审核之后的结果，供调用方确认本次最终是批准还是拒绝；本例的状态文字不代表真实外部发布。
    final: str


# 先确定任务类别，再把结果写回状态，后面的条件边据此选路。
def classify_node(state: State) -> dict[str, bool]:
    """用几个示范关键字决定要不要先走离线查询节点。"""
    # 把检索关键词中的英文统一为小写，避免大小写差异改变路由；中文字符保持原样。
    query = state["query"].lower()
    return {"needs_search": any(word in query for word in ("population", "weather", "latest", "人口", "天气", "最新"))}


# 将检索结果写入状态，后续生成节点需要实际读到这些资料。
def search_node(state: State) -> dict[str, str]:
    """从很小的离线资料表取值；这里不代表真正的搜索引擎。"""
    for key, value in {"taipei": "台北人口：约 260 万（固定教学数据）。", "台北": "台北人口：约 260 万（固定教学数据）。", "天气": "示例天气：晴，25 摄氏度。"}.items():
        if key in state["query"].lower():
            return {"search_result": value}
    return {"search_result": "离线资料库中没有匹配条目。"}


# 只决定下一条边，不在这个判断函数里执行检索。
def should_search(state: State) -> Literal["search", "respond"]:
    # 返回的是条件边要使用的目标名称；这里只选路，不执行检索，也不会生成回答。
    return "search" if state["needs_search"] else "respond"


def make_ollama_llm() -> ChatOpenAI:
    """用 OpenAI-compatible 接口连到本机 Ollama。"""
    # model：显式指定本次使用的模型或测试替身，避免悄悄使用默认值；换模型后要重新核对本例的输出约定。
    # base_url：把请求发到这个地址，而不是 SDK 的默认云服务；使用 OpenAI 客户端并不代表数据一定发给 OpenAI。
    # api_key：这是客户端要求的鉴权参数；本机 Ollama 的占位值不是云端密钥，也不提供执行授权。
    # temperature：降低采样随机性，便于比较实验结果；即使为 0，也不能保证每次输出完全相同或事实正确。
    return ChatOpenAI(model=MODEL, base_url="http://localhost:11434/v1", api_key="ollama", temperature=0)


# 把节点和条件边连接成图；阅读分支函数，确认什么情况下继续、暂停或结束。
def build_graph(llm: Any, checkpointer: Any | None = None) -> Any:
    """建立共用图；调用方可放入 Ollama 或 Anthropic chat model。"""
    def respond_node(state: State) -> dict[str, str]:
        # 直接回答路径不会产生 search_result，因此提供明确的缺省说明；不能为了缺字段伪造一条检索结果。
        context = state.get("search_result", "本次无需检索。")
        message = llm.invoke(f"请谨慎地用一句中文回答：{state['query']}\n离线资料：{context}")
        # 框架通常返回带 content 的消息对象，离线替身也可能直接返回文本；统一取正文并去除空白后才能判断是否真的有草稿。
        draft = str(getattr(message, "content", message)).strip()
        if not draft:
            raise RuntimeError("模型返回了空草稿。")
        return {"draft": draft}

    def review_node(state: State) -> dict[str, bool]:
        # 把待审草稿带到暂停事件中，等待人给出明确决定；恢复时该节点可能重新执行，暂停前不要放不可重复的外部动作。
        decision = interrupt({"action": "publish", "draft": state["draft"], "question": "是否批准这份草稿？"})
        # 只接受真正的布尔决定；字符串 "false" 在 Python 中也是真值，若直接判断真假会把拒绝误当作批准。
        if not isinstance(decision, bool):
            raise ValueError("人工审核必须用 true 或 false 恢复流程。")
        return {"approved": decision}

    def final_node(state: State) -> dict[str, str]:
        # 把批准与拒绝分支明确区分，供测试核对路由；这里仅生成教学状态文本，没有调用真实发布接口。
        prefix = "PUBLISHED" if state["approved"] else "REJECTED"
        # 把审核结论和对应草稿一起交还调用方，使人能确认被批准或拒绝的到底是哪一份内容。
        return {"final": f"{prefix}: {state['draft']}"}

    # 以 State 作为图中各节点共享的数据约定；每个节点返回自己的更新，数据通过状态传递而非隐式全局变量。
    graph = StateGraph(State)
    # 把 classify 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    graph.add_node("classify", classify_node)
    # 把 search 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    graph.add_node("search", search_node)
    # 把 respond 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    graph.add_node("respond", respond_node)
    # 把 review 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    graph.add_node("review", review_node)
    # 把 final 注册为一个可调度节点；后面的连线使用这个名称找到对应处理函数。
    graph.add_node("final", final_node)
    # 每次新任务先分类，避免所有问题都机械地检索。
    graph.add_edge(START, "classify")
    # 由路由函数返回目标名称，再通过这张映射选边；返回了未登记的名称就不能继续，路径必须可核对。
    graph.add_conditional_edges("classify", should_search, {"search": "search", "respond": "respond"})
    # 检索完成后必须把结果交给生成节点，保证草稿用的是实际资料。
    graph.add_edge("search", "respond")
    # 生成草稿后进入人工审核，不能直接把模型输出视为已获批准。
    graph.add_edge("respond", "review")
    # 只有审核节点取得决定后才进入最终分支，保持生成与授权分开。
    graph.add_edge("review", "final")
    # 批准或拒绝都在这里结束，避免完成之后意外继续循环。
    graph.add_edge("final", END)
    # 把节点与连线变成可运行的图；这一步是组装流程，还没有替用户执行一次任务。
    # checkpointer：把图的中间状态交给检查点保存器；暂停后恢复需要这份状态，单靠再次调用图无法找回进度。
    return graph.compile(checkpointer=checkpointer or InMemorySaver())


def run(query: str, approve: bool, llm: Any | None = None, thread_id: str = "stage4-demo") -> dict[str, Any]:
    """先跑到人工审核点，再用同一个 ``thread_id`` 恢复执行。"""
    graph = build_graph(llm or make_ollama_llm())
    # 运行配置必须在暂停和恢复时保持同一个会话标识，否则恢复会找不到刚才待审核的状态。
    # configurable：传递可在每次运行时指定的图配置；会话标识放在这里，供检查点系统定位历史状态。
    # thread_id：定位同一条工作流的检查点。暂停和恢复必须使用同一个值；换值通常会进入另一条会话。
    config = {"configurable": {"thread_id": thread_id}}
    # 第一次运行应停在人工审核点；此时只有待审草稿，不能声称已经发布。
    paused = graph.invoke({"query": query}, config=config)
    if "__interrupt__" not in paused:
        raise RuntimeError("发布前应先暂停并等待人工审核。")
    # 把明确的批准或拒绝传回同一工作流；后面的检查用实际恢复结果判断是否完成。
    resumed = graph.invoke(Command(resume=approve), config=config)
    if not resumed.get("final"):
        raise RuntimeError("恢复后未生成最终结果。")
    return resumed


if __name__ == "__main__":
    approved = run("台北人口是多少？", approve=True)
    # thread_id：定位同一条工作流的检查点。暂停和恢复必须使用同一个值；换值通常会进入另一条会话。
    rejected = run("请介绍 Python", approve=False, thread_id="stage4-reject")

## CodeAct 是怎么运作的

LLM 不回 JSON、而是**回 Python code block**：

````
（user）Find 台北人口, divide by NYC, give ratio.

（LLM 回应）
```python
pop_taipei = lookup_fact(query="台北人口") # 2602000
pop_nyc = lookup_fact(query="纽约人口") # 8336000
ratio = calculator(expression=f"{pop_taipei}/{pop_nyc}") # 0.3122
print(ratio)
```

（Smolagents 执行这段 code、把 print 结果接回去给 LLM 继续）
````

这份示例明确使用 Docker executor。Jupyter 控制端口只绑定到主机的 `127.0.0.1`，并移除 Linux capabilities、禁止提权与 pickle，再限制内存、process 数与 agent 步数。一般 Docker bridge **仍可能让容器连接外部网络或主机服务**，所以这只是受控教学示例，不是 production sandbox。需要执行不可信程序码时，还要加真正的 egress／host 防火墙，或改用有正式隔离边界的远程 sandbox。

## CodeAct vs JSON tool 对照

| 维度 | JSON tool | CodeAct |
|---|---|---|
| LLM 输出形式 | 结构化 JSON | Python 程序码 |
| 变数绑定 | LLM 要自己记得 / 重复调用 | 自然有 variable（`pop_taipei = ...`） |
| 多步运算 | 每步一次 LLM call | 一次写好几行 code |
| 一轮 token 数 | 较少 | 较多（code 较长） |
| 对小 model | 较友善（稳定的 JSON） | 较吃力（要产正确 Python） |
| Debug 友善 | tool call 看得清楚 | 看 code execution log |
| 安全考量 | allowlist + 参数验证 | 不可信程序码，必须隔离、限权、限资源 |
| 哪些题目擅长 | 单步、边界明确 | 多步运算、需要中间 variable |

**HuggingFace 的观点**：CodeAct 更贴近“人类怎么解问题”——你也是用变数记中间结果、不是每步都重新查。

## 常见坑

- **`@tool` 函数 docstring 是 prompt 的一部分**：Smolagents 把 docstring 当 tool description 给 LLM 看。**docstring 没写好、LLM 不知道何时用这 tool**
- **把 Docker 当完整 sandbox**：错。这份示例只缩小权限并把控制端口绑定到 loopback；上线前还要做镜像、权限、egress、host access、资源与记录审查
- **`max_steps` 不够**：先看错在哪一步，不要只把数字调大；较大上限会增加费用与 loop 风险
- **模型程序码有 syntax error**：Smolagents 可以把错误接回模型修正，但会增加步数；是否换模型要看评测结果

"""Stage 4 练习 4：比较有界 CodeAct 与安全的 JSON tool 算术路径。

JSON tool 只允许固定工具与固定参数；算式先转成 AST，再逐节点检查。
CodeAct 会执行模型产生的 Python，所以只在受限制的 Docker 容器里示范。
Jupyter 控制埠只绑到主机的 127.0.0.1；容器仍可能连网，因此这不是 production
sandbox。两条路都设有步数与输入边界。
"""

# 比较 JSON 参数调用与代码行动；算术用 AST 白名单，CodeAct 的执行依赖受限 Docker。

from __future__ import annotations

import ast
import os
import sys
from typing import Any

# 不同终端对标准输出对象的实现不同，先确认支持重设编码，再开启中文输出，避免测试替身上报属性错误。
if hasattr(sys.stdout, "reconfigure"):
    # 统一终端的中文编码；无法编码的字符只替换显示，不能让日志打印失败掩盖实际业务结果。
    sys.stdout.reconfigure(encoding="utf-8", errors="replace")

from smolagents import CodeAgent, OpenAIServerModel, ToolCallingAgent, tool

MODEL = os.environ.get("MODEL", "qwen2.5:3b")
OLLAMA_BASE = os.environ.get("OLLAMA_API_BASE", "http://localhost:11434/v1")
MAX_AST_NODES = 40
MAX_NUMBER = 1_000_000


# 解析表达式后只允许约定的算术节点，并限制数值与表达式规模。
def evaluate_arithmetic(expression: str) -> float:
    """只计算小型算术 AST；模型文字不会交给 ``eval`` 或 ``exec``。"""
    if not isinstance(expression, str) or len(expression) > 120:
        raise ValueError("表达式必须是最多 120 字符的字符串。")
    # 只把模型源码解析为语法树，不运行它；后面还要逐节点白名单检查，解析成功不代表安全。
    tree = ast.parse(expression, mode="eval")
    if sum(1 for _ in ast.walk(tree)) > MAX_AST_NODES:
        raise ValueError("表达式过于复杂。")

    def visit(node: ast.AST) -> float:
        if isinstance(node, ast.Expression):
            return visit(node.body)
        if isinstance(node, ast.Constant) and type(node.value) in (int, float):
            value = float(node.value)
            if abs(value) > MAX_NUMBER:
                raise ValueError("数值的绝对值超出允许范围。")
            return value
        if isinstance(node, ast.UnaryOp) and isinstance(node.op, (ast.UAdd, ast.USub)):
            return visit(node.operand) if isinstance(node.op, ast.UAdd) else -visit(node.operand)
        if isinstance(node, ast.BinOp) and isinstance(node.op, (ast.Add, ast.Sub, ast.Mult, ast.Div)):
            left, right = visit(node.left), visit(node.right)
            if isinstance(node.op, ast.Add): return left + right
            if isinstance(node.op, ast.Sub): return left - right
            if isinstance(node.op, ast.Mult): return left * right
            if right == 0: raise ValueError("除数不能为零。")
            return left / right
        raise ValueError(f"不支持的算术语法：{type(node).__name__}")

    value = visit(tree)
    if abs(value) > MAX_NUMBER:
        raise ValueError("计算结果的绝对值超出允许范围。")
    return value


@tool
def calculator(expression: str) -> str:
    """在长度和数值范围限制内计算一个简单算术表达式。

    Args:
        expression: 只允许数字、括号，以及 +、-、*、/ 四种运算符。
    """
    try:
        return str(evaluate_arithmetic(expression))
    except (SyntaxError, ValueError) as error:
        return f"错误：{error}"


@tool
def lookup_fact(query: str) -> str:
    """只从固定白名单返回一项离线事实。

    Args:
        query: Either ``台北人口`` or ``纽约人口``.
    """
    facts = {'台北人口': "2602000", '纽约人口': "8336000"}
    return facts.get(query.strip().lower(), f"未知条目：{query}")


# 先验证工具名称和参数，再执行对应函数；模型字符串不会被当 Python 程序运行。
def run_json_tool_call(call: dict[str, Any]) -> str:
    """JSON tool 边界：工具名、参数形状与算式都要先通过检查。"""
    if not isinstance(call, dict) or set(call) != {"name", "arguments"}:
        raise ValueError("工具请求必须且只能包含 name 和 arguments。")
    if call["name"] != "calculator" or not isinstance(call["arguments"], dict) or set(call["arguments"]) != {"expression"}:
        raise ValueError("只允许调用 calculator，并传入 expression 参数。")
    return calculator.forward(expression=call["arguments"]["expression"])


# Docker 控制通道与容器执行权限分开设置；有配置仍要通过真实容器测试。
def codeact_executor_config() -> dict[str, Any]:
    """缩小示范容器权限，并把 Jupyter 控制埠只绑到本机。"""
    return {
        "allow_pickle": False,
        "host": "127.0.0.1",
        "container_run_kwargs": {
            "network_mode": "bridge",
            "mem_limit": "256m",
            "pids_limit": 64,
            "cap_drop": ["ALL"],
            "security_opt": ["no-new-privileges"],
        },
    }


# 把模型接到 CodeAgent 和受限执行器，运行前需要对应 Docker 环境。
def build_codeact_agent(model: Any | None = None, executor_type: str = "docker") -> CodeAgent:
    """建立最多走四步、且预设只在 Docker 里执行程序码的 agent。"""
    # api_key：这是客户端要求的鉴权参数；本机 Ollama 的占位值不是云端密钥，也不提供执行授权。
    model = model or OpenAIServerModel(model_id=MODEL, api_base=OLLAMA_BASE, api_key="ollama")
    executor_kwargs = codeact_executor_config() if executor_type == "docker" else {}
    # tools：只向模型公开本轮需要的工具说明。模型提出调用后，执行层仍须检查名称、参数与权限。
    # model：显式指定本次使用的模型或测试替身，避免悄悄使用默认值；换模型后要重新核对本例的输出约定。
    # max_steps：限制循环最多执行多少步；它是程序的停止条件，不能只在提示词里写“不要无限循环”。
    return CodeAgent(tools=[calculator, lookup_fact], model=model, max_steps=4, additional_authorized_imports=[], executor_type=executor_type, executor_kwargs=executor_kwargs)


# 模型通过结构化工具请求行动，工具内部仍负责输入检查。
def build_json_agent(model: Any) -> ToolCallingAgent:

## 为什么 type-safe agent 重要

```
Stage 3 练习 6：schema = JSON Schema in prompt
    LLM 看到、但回什么是 LLM 决定（可能违反）

Stage 4 练习 5：schema = Pydantic model in code
    LLM 违反 → framework 自动 raise → retry / 修
    成功回传的 output 已通过 runtime 格式检查
    内容是否正确仍要另外验证
```

对 production：

| 需求 | 纯 prompt schema | Pydantic AI |
|---|---|---|
| LLM 偶尔少栏位 | 你的下游 code 要 try/except | 自动 retry 直到符合 |
| 型别错（confidence="high"） | 下游 crash | Pydantic ValidationError、retry |
| 边界错（confidence=1.5） | 下游用错误值 | 拒绝、retry |
| 多余栏位 | 依你的 parser 而定 | 依 Pydantic model 设定处理 |

**结论**：下游程序需要固定栏位时，typed output 很有用。Stage 3 练习 6 教 schema 设计；这题把 schema 变成 runtime contract，再提醒你补上事实查核。

## Pydantic AI 核心观念

### Agent + output_type

```python
agent = Agent(
    model=...,
    output_type=AnswerWithConfidence, # ← 强制 LLM 回这个 shape
    system_prompt="..."
)
result = agent.run_sync(question)
answer: AnswerWithConfidence = result.output # 已验证的物件
```

**重点**：framework 把 Pydantic schema 转成 structured output 指示，执行 validation，失败时依设定重试。重试成功只表示格式合格。

### Field constraints

```python
confidence: float = Field(ge=0.0, le=1.0, description="...")
```

`ge` / `le` 是 Pydantic 的 numeric bound。LLM 回 `1.5` 会被 ValidationError 挡下、retry。

### 自动 retry

```python
Agent(..., retries=3) # default 1，可调
```

Pydantic AI 看到 ValidationError、会把错误讯息塞回 prompt、要求 LLM 重产。

## 常见坑

- **`output_type` 太复杂**：nested model 越深越难产生与维护。先用最少必要栏位，再用评测决定是否拆分
- **缺 `description`**：`Field(...)` 没写 `description=`、LLM 看不到栏位用途、易误填
- **`retries=0`**：失败就 raise。重试次数要依费用、延迟与失败模式设定，并保留上限
- **小 model + 深 nested**：qwen2.5:3b 可能 retry 多次仍不对。换大 model 或扁平 schema

"""Stage 4 练习 5：用 Pydantic AI ``output_type`` 检查输出形状。

型别检查像表格的格子：它能确保栏位存在、种类正确、数字在范围内，
但不能证明内容是真的。真实性仍要靠可信来源与额外的语义检查。
"""

# 检查字段类型、范围与非空约束；结构合法不代表答案或置信度在事实层面正确。

from __future__ import annotations

import os
import sys
from typing import Any

# 不同终端对标准输出对象的实现不同，先确认支持重设编码，再开启中文输出，避免测试替身上报属性错误。
if hasattr(sys.stdout, "reconfigure"):
    # 统一终端的中文编码；无法编码的字符只替换显示，不能让日志打印失败掩盖实际业务结果。
    sys.stdout.reconfigure(encoding="utf-8", errors="replace")

# Field 把长度、范围等约束附在字段上；格式通过之后，还要核对业务规则和事实依据。
from pydantic import BaseModel, Field, field_validator
from pydantic_ai import Agent
from pydantic_ai.models.openai import OpenAIChatModel
from pydantic_ai.providers.openai import OpenAIProvider

MODEL = os.environ.get("MODEL", "qwen2.5:3b")
OLLAMA_BASE = os.environ.get("OLLAMA_API_BASE", "http://localhost:11434/v1")


# Pydantic 限制字段形状与范围，confidence 不表示经过统计校准的正确率。
class AnswerWithConfidence(BaseModel):
    """已验证的资料形状；它不是答案为真的证据。"""
    # answer：保存最终对外文本；非空只说明模型有输出，事实与引用是否正确仍要独立核对。
    # min_length：要求至少有指定数量的字符或元素，避免空答案、空来源列表看起来结构合法却无法使用。
    answer: str = Field(min_length=1, description="非空且不夸大确定性的中文答案。")
    # confidence：这里只约束模型自报值的范围，不能把它当成经过校准的真实正确率。
    # ge：设置包含边界的数值下限；类型正确但越界的值也必须被拒绝。
    # le：设置包含边界的数值上限，防止过大置信度或预算通过类型检查。
    confidence: float = Field(ge=0.0, le=1.0, description="置信度必须在 0.0 到 1.0 之间。")
    # sources：要求提供可回查的来源标签，让审核者能核对依据；标签存在不等于来源真的支持断言。
    # min_length：要求至少有指定数量的字符或元素，避免空答案、空来源列表看起来结构合法却无法使用。
    sources: list[str] = Field(min_length=1, description="至少提供一个非空的来源标签。")

    @field_validator("answer")
    @classmethod
    def answer_is_not_blank(cls, value: str) -> str:
        if not value.strip():
            raise ValueError("answer 不能为空白")
        return value.strip()

    @field_validator("sources")
    @classmethod
    def sources_are_nonempty_labels(cls, value: list[str]) -> list[str]:
        if not all(isinstance(item, str) and item.strip() for item in value):
            raise ValueError("sources 中的来源标签必须是非空字符串")
        return [item.strip() for item in value]


def make_ollama_model() -> OpenAIChatModel:
    """用目前 Pydantic AI 的 OpenAIChatModel 连到本机 Ollama。"""
    # base_url：把请求发到这个地址，而不是 SDK 的默认云服务；使用 OpenAI 客户端并不代表数据一定发给 OpenAI。
    # api_key：这是客户端要求的鉴权参数；本机 Ollama 的占位值不是云端密钥，也不提供执行授权。
    return OpenAIChatModel(MODEL, provider=OpenAIProvider(base_url=OLLAMA_BASE, api_key="ollama"))


def build_agent(model: Any | None = None) -> Agent:
    """建立会重试两次、并要求 AnswerWithConfidence 的 agent。"""
    return Agent(
        model or make_ollama_model(),
        # output_type：把最终答案约束为这个类型，尽早暴露缺字段或类型错误；仍需另做事实核对。
        output_type=AnswerWithConfidence,
        retries=2,
        instructions=(
            "返回 AnswerWithConfidence：给出非空中文答案、0 到 1 之间的置信度，"
            "以及至少一个来源标签。结构合法不能证明内容真实。"
        ),
    )


def run(question: str, model: Any | None = None) -> AnswerWithConfidence:
    """执行 agent，并在返回前再检查本练习的结果契约。"""
