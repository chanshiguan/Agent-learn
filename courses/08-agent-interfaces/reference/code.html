"""真实 Playwright 执行器 + 固定动作提议；不是模型智能效果测试。"""

# 默认打开内存中的演示页面；live 才访问 example.com，观察结果与截图来自真实浏览器。
import argparse
import json
from pathlib import Path
from playwright.sync_api import sync_playwright
from policy import check


def run(output, live=False):
    output.mkdir(parents=True, exist_ok=True)
    trace = []
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=True)
        context = browser.new_context(accept_downloads=False)
        # 每条网络请求核对域名；本例不接受第三方子资源。
        context.route("**/*", lambda route: route.continue_() if check("read", route.request.url) == "ALLOW" else route.abort())
        page = context.new_page()
        if live:
            # timeout：限制当前等待操作的时长，避免它一直阻塞；这个局部超时不等于整个 Agent 的总时限。
            page.goto("https://example.com", timeout=15000)
            if check("read", page.url) != "ALLOW":
                raise RuntimeError("最终 URL 越界")
        else:
            page.set_content('<html><head><title>课程演示</title></head><body><h1>虚构页面</h1><p>没有登录或发送功能。</p></body></html>')
        proposals = [{"action": "read", "url": "https://example.com"}, {"action": "send", "url": "https://example.com"}, {"action": "read", "url": "https://evil.test"}]
        for proposal in proposals:
            verdict = check(**proposal)
            record = {**proposal, "policy": verdict, "executed": False}
            if verdict == "ALLOW":
                record.update({"title": page.title(), "url_observed": page.url, "executed": True})
                page.screenshot(path=str(output / "observed.png"), full_page=True)
            else:
                record["reason"] = "需可信用户批准，当前演示停止该动作" if verdict == "ASK" else "策略拒绝"
            trace.append(record)
        browser.close()
    # 将当前产物保存到明确文件，供另一次运行或人工复核；保存成功本身不代表内容已通过验收。
    # 以 JSON 保留字段和类型，便于重跑比较或人工核对；不能只保存对象的调试显示。
    # ensure_ascii：设为 False 时直接保留中文，方便阅读日志与工具结果；JSON 结构和字段含义不变。
    # indent：给本地 JSON 产物增加缩进，方便人工逐字段检查；不会改变数据内容。
    (output / "trace.json").write_text(json.dumps({"mode": "真实浏览器/固定提议", "network": live, "steps": trace}, ensure_ascii=False, indent=2))
    return trace


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    # 为运行者提供明确的选项与默认值；这里的限制让同一实验可以按相同条件重跑。
    parser.add_argument("--live", action="store_true", help="访问 example.com；默认只用内存页面")
    # 为运行者提供明确的选项与默认值；这里的限制让同一实验可以按相同条件重跑。
    parser.add_argument("--output", type=Path, default=Path("/tmp/ai-learn-browser"))
    # 先把命令行输入转为受约束的参数，防止错误模式或缺少参数到执行一半才暴露。
    args = parser.parse_args()
    # 以 JSON 保留字段和类型，便于重跑比较或人工核对；不能只保存对象的调试显示。
    # ensure_ascii：设为 False 时直接保留中文，方便阅读日志与工具结果；JSON 结构和字段含义不变。
    # indent：给本地 JSON 产物增加缩进，方便人工逐字段检查；不会改变数据内容。

"""本地虚构表单：观察→提议→策略→人工决定→执行→外部状态验证。无发送。"""

# 先读 observe/propose/policy，再看 run；固定提议用于隔离执行器问题，不代表模型规划。
import argparse
import json
from pathlib import Path
from playwright.sync_api import sync_playwright


# 读取页面真实值和版本，既用于执行前检查，也用于执行后验收。
def observe(page):
    return {'title':page.title(),'draft':page.locator('#draft').input_value(),
            'version':page.locator('#version').inner_text()}


# 按场景返回固定请求；请求携带观察快照，便于发现批准后页面变化。
def propose(observation, scenario):
    if scenario == 'offsite':
        return {'type':'navigate','url':'https://evil.test','expected':observation}
    if scenario == 'send':
        return {'type':'send','expected':observation}
    return {'type':'fill','value':'课程学习草稿','expected':observation}


# 填写需要操作者批准；发送和跳转都不属于本题允许的动作。
def policy(action):
    if action['type'] == 'fill':
        return 'ASK'
    return 'BLOCK'


# 只有策略允许、操作者接受、页面未变三项都满足，才调用浏览器填写。
def run(output, decision='reject', scenario='normal'):
    output=Path(output); output.mkdir(parents=True,exist_ok=True)
    with sync_playwright() as p:
        browser=p.chromium.launch(headless=True)
        page=browser.new_page()
        # 演示页不需要网络，所有页面请求都阻断。
        page.route('**/*',lambda route:route.abort())
        page.set_content('<title>本地虚构表单</title><label>草稿<input id="draft"></label><span id="version">1</span>')
        before=observe(page)
        action=propose(before,scenario)
        verdict=policy(action)
        record={'before':before,'proposal':action,'policy':verdict,'executed':False}
        if verdict == 'BLOCK':
            record['status']='BLOCKED'
        else:
            # 只允许操作者通过 CLI 明确选择；提议无法自行给出批准。
            record['decision']=decision
            if decision != 'accept':
                record['status']='REJECTED'
            else:
                if scenario == 'stale':
                    page.locator('#version').evaluate('(e) => e.textContent = "2"')
                # 批准后再读一次当前页面；状态变化就停止，旧批准不能用于新状态。
                if observe(page) != action['expected']:
                    record['status']='STALE_REQUIRES_NEW_REVIEW'
                else:
                    try:
                        if scenario == 'failure':
                            raise RuntimeError('教学注入：执行器不可用')
                        page.locator('#draft').fill(action['value'])
                        record['executed']=True
                        # 执行后独立读取输入框，核对真正写进去的内容，而不是相信执行器口头报告。
                        after=observe(page)
                        if after['draft'] != action['value']:
                            raise RuntimeError('实际页面值与提议不一致')
                        record['status']='VERIFIED'
                    except Exception as error:
                        record['status']='FAILED'
                        record['error']=str(error)
        record['after']=observe(page)
        page.screenshot(path=str(output/'state.png'),full_page=True)
        browser.close()
    # 将当前产物保存到明确文件，供另一次运行或人工复核；保存成功本身不代表内容已通过验收。
    # 以 JSON 保留字段和类型，便于重跑比较或人工核对；不能只保存对象的调试显示。
    # ensure_ascii：设为 False 时直接保留中文，方便阅读日志与工具结果；JSON 结构和字段含义不变。
    # indent：给本地 JSON 产物增加缩进，方便人工逐字段检查；不会改变数据内容。
    (output/'receipt.json').write_text(json.dumps(record,ensure_ascii=False,indent=2))
    return record


if __name__ == '__main__':
    p=argparse.ArgumentParser()
    # 为运行者提供明确的选项与默认值；这里的限制让同一实验可以按相同条件重跑。
    p.add_argument('--decision',choices=['accept','reject'],default='reject')
    # 为运行者提供明确的选项与默认值；这里的限制让同一实验可以按相同条件重跑。
    p.add_argument('--scenario',choices=['normal','offsite','send','stale','failure'],default='normal')
    # 为运行者提供明确的选项与默认值；这里的限制让同一实验可以按相同条件重跑。
    p.add_argument('--output',type=Path,default=Path('/tmp/ai-learn-action-loop'))
    # 先把命令行输入转为受约束的参数，防止错误模式或缺少参数到执行一半才暴露。
    a=p.parse_args()
    # 以 JSON 保留字段和类型，便于重跑比较或人工核对；不能只保存对象的调试显示。
    # ensure_ascii：设为 False 时直接保留中文，方便阅读日志与工具结果；JSON 结构和字段含义不变。
    # indent：给本地 JSON 产物增加缩进，方便人工逐字段检查；不会改变数据内容。
    print(json.dumps(run(a.output,a.decision,a.scenario),ensure_ascii=False,indent=2))

"""沙箱中的固定绘图任务；读取两列 CSV，输出独立 SVG。"""

# 读取固定 CSV 后输出 SVG；本脚本负责绘图，隔离依靠运行它的容器配置。
import csv
from html import escape
from pathlib import Path


def plot(source, destination):
    # encoding：明确按 UTF-8 读写中文，避免不同终端或系统默认编码导致乱码。
    with source.open(encoding="utf-8", newline="") as f:
        rows = list(csv.DictReader(f))
    if not rows or len(rows) > 20:
        raise ValueError("需要 1–20 条记录")
    values = [float(row["value"]) for row in rows]
    if any(not 0 <= value <= 10000 for value in values):
        raise ValueError("数值必须在 0–10000")
    scale = 360 / max(max(values), 1)
    parts = ['<svg xmlns="http://www.w3.org/2000/svg" width="600" height="700"><rect width="100%" height="100%" fill="white"/>']
    for i, (row, value) in enumerate(zip(rows, values)):
        y = 20 + i * 32
        parts.append(f'<text x="10" y="{y+15}">{escape(row["label"])}</text><rect x="150" y="{y}" width="{value*scale}" height="22" fill="#9a3412"/><text x="520" y="{y+15}">{value:g}</text>')
    parts.append('</svg>')
    destination.parent.mkdir(parents=True, exist_ok=True)
    # 将当前产物保存到明确文件，供另一次运行或人工复核；保存成功本身不代表内容已通过验收。
    destination.write_text(''.join(parts))


if __name__ == "__main__":
    plot(Path("/input/data.csv"), Path("/output/chart.svg"))

"""动作策略检查器；不是网络隔离或完整浏览器沙箱。"""

# 先解析 URL 并核对精确主机，再按动作分类；ASK 只表示待批准，不会执行操作。
from urllib.parse import urlsplit


def check(action, url):
    try:
        parts = urlsplit(url)
        port = parts.port
    except (ValueError, TypeError):
        return "BLOCK"
    if parts.scheme != "https" or parts.hostname != "example.com" or port not in {None, 443}:
        return "BLOCK"
    if parts.username is not None or parts.password is not None:
        return "BLOCK"
    if not isinstance(action, str):
        return "BLOCK"
    name = action.strip().lower()
    if name in {"read", "screenshot"}:
        return "ALLOW"
    if name in {"login", "purchase", "delete", "send"}:
        return "ASK"
    return "BLOCK"


if __name__ == "__main__":
    for action, url in [("read", "https://example.com"), ("send", "https://example.com"), ("read", "https://example.com.evil.test"), ("read", "file:///etc/passwd")]:
        print(action, url, check(action, url))
