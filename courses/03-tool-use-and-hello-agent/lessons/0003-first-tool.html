01_hello_tool.py · 完整文件
"""练习 1：演示模型调用一个工具并根据工具结果生成最终回答。"""

# 导入 Python 标准库 json，用于把字典转换为便于打印和传输的 JSON 字符串。
import json

# 导入系统提示词；它负责告诉模型自己的角色以及应如何使用工具。
from agent import SYSTEM
# 导入模型请求函数的工厂；make_ask 会创建一个连接本机 Ollama 的请求函数。
from model_client import make_ask
# 导入工具定义 TOOLS，以及负责安全执行工具的 execute 函数。
from tools import TOOLS, execute


# 把示例的完整流程放进 main，避免该文件被其他模块导入时自动发起模型请求。
def main():
    # 本练习只把 TOOLS 中的第一个工具 get_weather 提供给模型，保持示例简单。
    tools = [TOOLS[0]]

    # 创建模型请求函数；这里只完成客户端配置，还没有向模型发送消息。
    ask = make_ask()

    # 建立首轮对话历史，模型每一轮都会看到这个列表中的全部消息。
    # 这份历史同时保存问题、模型请求和工具观察；下一轮需要它们才能沿用真实结果，保持调用与结果的顺序。
    messages = [
        # system 消息提供最高层的行为说明，让模型知道工具调用规则。
        {"role": "system", "content": SYSTEM},
        # user 消息是本次练习实际要模型处理的问题。
        {"role": "user", "content": "查询台北的示例气温。"},
    ]

    # 第一次请求模型：期望模型识别出需要天气数据，并返回 get_weather 工具调用。
    first = ask(messages, tools)

    # 首轮只能正常结束或请求工具；其他原因通常表示输出异常或达到长度上限。
    if first["finish_reason"] not in ("stop", "tool_calls"):
        # 立即终止，防止继续处理一个不完整的模型响应。
        raise RuntimeError("首轮输出异常或被截断")

    # 取出标准化后的 assistant 消息，里面可能包含文本或 tool_calls。
    assistant = first["message"]

    # 读取工具调用列表；模型没有返回 tool_calls 时，用空列表代替 None。
    # 没有调用时才进入最终回答分支；空列表是“本轮不执行工具”，不是已经取得工具结果。
    calls = assistant.get("tool_calls") or []

    # 本练习只支持一次、一个工具调用，并要求调用带有可关联结果的唯一 ID。
    if len(calls) != 1 or not calls[0].get("id"):
        # 响应结构不符合教学预期时给出明确错误，而不是在后续索引时报模糊错误。
        raise RuntimeError("本练习要求恰好一个带 ID 的工具请求；检查模型、schema 和原始响应")

    # 把模型的工具调用消息加入历史；第二轮模型需要看到自己刚才发起的调用。
    messages.append(assistant)

    # 取出唯一的工具调用，供后续打印、执行以及关联工具结果使用。
    call = calls[0]

    # 打印模型提出的工具名和参数；ensure_ascii=False 让中文保持可读。
    # 以 JSON 保留字段和类型，便于重跑比较或人工核对；不能只保存对象的调试显示。
    # ensure_ascii：设为 False 时直接保留中文，方便阅读日志与工具结果；JSON 结构和字段含义不变。
    print("工具请求：", json.dumps(call, ensure_ascii=False))

    # 执行模型请求的工具；集合 {"get_weather"} 明确限制本练习只授权天气工具。
    result = execute(
        # 模型请求调用的函数名，例如 get_weather。
        call["function"]["name"],
        # 模型生成的函数参数，通常是 JSON 字符串，例如 {"city": "台北"}。
        call["function"]["arguments"],
        # 本轮允许执行的工具白名单，防止模型调用未授权函数。
        {"get_weather"},
    )

    # 打印本地工具的执行结果，方便观察“模型请求—程序执行”的中间过程。
    # 以 JSON 保留字段和类型，便于重跑比较或人工核对；不能只保存对象的调试显示。
    # ensure_ascii：设为 False 时直接保留中文，方便阅读日志与工具结果；JSON 结构和字段含义不变。
    print("工具结果：", json.dumps(result, ensure_ascii=False))

    # 把工具结果作为一条 tool 消息追加到对话历史，供模型生成最终答案。
    messages.append(
        {
            # role=tool 表明这不是用户或模型文本，而是一次工具执行结果。
            "role": "tool",
            # 用同一个调用 ID 把结果和模型之前发起的工具请求对应起来。
            "tool_call_id": call["id"],
            # API 的消息内容使用字符串，因此把结果字典序列化为 JSON。
            # 把结果编码成协议可接受的 JSON；模型收到的是工具数据，不能把其中的指令当作系统授权。
            # ensure_ascii：设为 False 时直接保留中文，方便阅读日志与工具结果；JSON 结构和字段含义不变。
            "content": json.dumps(result, ensure_ascii=False),
        }
    )

    # 第二次请求模型：此时模型已看到工具结果，应据此生成自然语言答案。
    final = ask(messages, tools)

    # 固定两轮的入门示例要求第二轮直接结束，不能再次请求其他工具。
    if final["finish_reason"] != "stop" or final["message"].get("tool_calls"):
        # 若仍需调用工具，应升级为循环式 Agent，而不是继续硬编码第三、第四轮。
        raise RuntimeError("第二轮没有结束；需要通用 Agent Loop，而不是继续增加固定请求")

    # 从最终 assistant 消息中提取给用户看的文字内容。
    answer = final["message"].get("content")

    # 空字符串或 None 都不是有效答案，因此在这里显式失败。
    if not answer:
        # 明确指出问题来自最终答案为空，方便定位模型兼容性问题。
        raise RuntimeError("最终答案为空")

    # 输出模型结合天气工具结果生成的最终回答。
    print("最终答案：", answer)


# 只有直接运行本文件时 __name__ 才等于 "__main__"；被导入时不会进入此分支。
if __name__ == "__main__":
    # 启动上面定义的完整示例流程。
    main()

4. 完整模型连接代码

主线必读

本文件只负责“发请求、读响应、处理可重试网络错误”。业务循环留在 agent.py，工具执行留在 tools.py。这样出错时可以分别定位，而不是把所有异常都归因于 Prompt。

model_client.py · 完整文件
"""创建连接本机 Ollama 的模型请求函数，不读取或使用 OpenAI 云端密钥。"""

# 导入 os，用于读取可选的 OLLAMA_MODEL 环境变量。
import os
# 导入 time，用于在可重试错误发生后短暂等待。
import time


# 这是一个函数工厂：先配置客户端和模型，再返回真正发送请求的 ask 函数。
def make_ask():
    # 延迟导入 SDK；只有实际创建请求函数时才要求安装 openai 包。
    from openai import APIConnectionError, APIStatusError, OpenAI

    # 创建兼容 OpenAI API 的客户端，但把请求地址指向本机 Ollama。
    client = OpenAI(
        # Ollama 默认监听 11434 端口，并在 /v1 提供 OpenAI 兼容接口。
        # base_url：把请求发到这个地址，而不是 SDK 的默认云服务；使用 OpenAI 客户端并不代表数据一定发给 OpenAI。
        base_url="http://localhost:11434/v1",
        # OpenAI SDK 要求提供 api_key；本地 Ollama 默认不会验证这个占位值。
        # api_key：这是客户端要求的鉴权参数；本机 Ollama 的占位值不是云端密钥，也不提供执行授权。
        api_key="ollama",
        # 单次请求最多等待 60 秒，防止模型无响应时程序永久卡住。
        # timeout：限制当前等待操作的时长，避免它一直阻塞；这个局部超时不等于整个 Agent 的总时限。
        timeout=60.0,
        # 关闭 SDK 内置重试，由下面的代码统一控制重试次数。
        # max_retries：控制客户端自动重试次数。设为 0 可把重试统一交给外层，避免两层重试相乘。
        max_retries=0,
    )

    # 优先使用环境变量指定的模型；未设置时默认使用本课程的 gemma4:e4b。
    model = os.environ.get("OLLAMA_MODEL", "gemma4:e4b")

    # 定义真正发送请求的内部函数；它会记住上面的 client 和 model。
    def ask(messages, tools):
        # 最多尝试两次：第一次正常请求，发生可恢复错误时再重试一次。
        for attempt in range(2):
            # 捕获连接错误和部分临时 HTTP 错误，以决定是否重试。
            try:
                # 调用 Chat Completions 接口，把对话历史和工具定义一起交给模型。
                response = client.chat.completions.create(
                    # 指定本次请求使用的 Ollama 模型。
                    # model：显式指定本次使用的模型或测试替身，避免悄悄使用默认值；换模型后要重新核对本例的输出约定。
                    model=model,
                    # 传入 system、user、assistant、tool 等完整对话历史。
                    # messages：传入本轮模型要读取的角色与内容；工具循环还须保留历史请求和结果，模型才知道前一步实际发生了什么。
                    messages=messages,
                    # 告诉模型当前允许请求哪些函数工具。
                    # tools：只向模型公开本轮需要的工具说明。模型提出调用后，执行层仍须检查名称、参数与权限。
                    tools=tools,
                    # 限制模型本次最多生成 768 个输出 token，避免无边界生成。
                    # max_tokens：限制本次响应最多生成多少 token，约束延迟和输出规模；达到上限后必须检查是否截断。
                    max_tokens=768,
                    # 使用确定性更高的输出，便于课程示例稳定复现。
                    # temperature：降低采样随机性，便于比较实验结果；即使为 0，也不能保证每次输出完全相同或事实正确。
                    temperature=0,
                    # reasoning_effort：配置模型支持的推理行为；这里的值必须与所用服务兼容，不能当作所有模型通用的开关。
                    reasoning_effort="none",
                )

                # 正常响应至少要包含一个候选结果，否则无法继续处理。
                if not response.choices:
                    # 用明确异常代替后续访问 choices[0] 时产生的索引错误。
                    raise RuntimeError("API 没有返回 choices")

                # 本课程只使用第一个候选回答。
                choice = response.choices[0]

                # SDK 可能不返回 usage；缺失时用 0，保证返回结构稳定。
                # 用量来自服务实际报告，用于成本和预算核对；缺失时保留未知，不自行估算成确定账单。
                usage = response.usage.total_tokens if response.usage else 0

                # 把 SDK 对象整理为课程其他文件容易处理的普通字典。
                return {
                    # 删除值为 None 的字段，保留文本内容或工具调用等有效字段。
                    # 把 SDK 或模型对象转成普通数据，方便保存与返回；不要把对象的调试字符串当作协议 JSON。
                    # exclude_none：省略值为 None 的可选字段，避免把“未返回”误写成一项有内容的协议字段。
                    "message": choice.message.model_dump(exclude_none=True),
                    # 保存结束原因，用于区分正常回答、工具调用和内容截断。
                    "finish_reason": choice.finish_reason,
                    # 保存本次请求的总 token 数，便于后续观察用量。
                    "tokens": usage,
                }

            # 连接不上本机 Ollama 时进入这里，例如服务尚未启动。
            except APIConnectionError:
                # attempt 从 0 开始；等于 1 表示第二次尝试也失败了。
                if attempt == 1:
                    # 原样抛出 SDK 异常，让调用者看到真实的连接失败原因。
                    raise

            # Ollama 返回非成功 HTTP 状态码时进入这里，并保留错误详情。
            except APIStatusError as error:
                # 第二次失败，或错误本身不可恢复时，不再继续重试。
                if attempt == 1 or error.status_code not in (429, 500, 502, 503, 504):
                    # 429 和常见 5xx 可重试；其他状态码通常需要修改请求后再试。
                    raise

            # 第一次遇到可重试错误后等待半秒，避免立刻再次冲击服务。
            time.sleep(0.5)

        # 理论上循环只会成功返回或抛出异常；此处作为防御性兜底。
        raise RuntimeError("模型请求未完成")

    # 返回内部函数而不是立刻请求模型，让调用者可以多次复用同一客户端配置。
    return ask

5. 按消息历史理解代码

主线必读
顺序role内容责任
1system固定任务与边界程序提供
2user查询台北示例气温用户提供
3assistant请求 get_weather，ID=c1模型提出
4tooltool_call_id=c1，结果为 26°C程序执行后提供
5assistant据结果生成中文回答模型生成

messages.append(assistant) 保存模型发出的原请求；下一条工具结果才有上下文。只返回工具结果，却丢掉 assistant 请求，可能导致协议错误，或让模型看不懂这个结果在回答哪一次调用。

tool_call_id 应原样使用模型返回的 ID。工具名称不能替代 ID：同一个函数可能一次被调用两次，分别查询不同城市。

ensure_ascii=False 让中文保持可读，不改变语义；model_dump(exclude_none=True) 把 SDK 对象转成可回传的字典，并省掉空字段。

6. 为什么请求两次

主线必读

第一轮模型还没得到工具结果，只能请求执行。第二轮才看见程序查到的数据。在本课的客户端工具流程中，一次工具往返通常需要这两轮模型请求；后续若还有工具请求，则交给下一单元的循环。

工具请求出现时，message.content 可能为空，这是正常现象。要查看 tool_calls，而不是只打印 content。usage 是 token 统计，也不是最终回答；最终文字读取 message.content。

7. 单工具最小程序的边界

主线必读

它故意要求首轮恰好一个工具请求，第二轮直接结束。若模型首轮直接回答，程序报错提醒没有完成实验；若第二轮继续要工具，程序也报错。固定写两轮适合认识协议，不能代替通用循环。

如果模型没有调用工具，记录原始返回、finish_reason、工具定义和模型版本。保持同题同 Schema 重试三次，看是偶发失误还是稳定失败；不要把 JSON 文字中的伪调用当成原生 tool_calls。
