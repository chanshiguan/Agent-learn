"""可复现的故障注入与删除实验：固定程序输出，不冒充模型评测。"""

# 每次只改变是否保留来源检查，用同一套正反例观察退步；不测模型能力。
import json

CASES=[{'id':'normal','refs':['d1'],'expected':'ok'},
       {'id':'bad_ref','refs':['invented'],'expected':'stop'},
       {'id':'empty','refs':[],'expected':'stop'},
       {'id':'timeout','refs':['d1'],'error':'timeout','expected':'stop'}]


# 本题既检查返回值，也检查异常是否符合预期；正确停止也可能是通过。
def trial(case, check_sources):
    if case.get('error'):
        return 'stop'
    if check_sources and (not case['refs'] or any(r!='d1' for r in case['refs'])):
        return 'stop'
    return 'ok'


# 逐题保存基线与删除后的差异，避免只看正常题掩盖故障。
def experiment():
    rows=[]
    for variant,check in [('baseline',True),('remove_source_check',False)]:
        for case in CASES:
            actual=trial(case,check)
            rows.append({'variant':variant,'case':case['id'],'actual':actual,'passed':actual==case['expected']})
    return {'mode':'确定性故障实验','rows':rows,'decision':'保留来源检查：删除后 bad_ref 与 empty 回退。该控制不是可随模型升级删除的措辞补丁。'}


if __name__=='__main__':
    # 以 JSON 保留字段和类型，便于重跑比较或人工核对；不能只保存对象的调试显示。
    # ensure_ascii：设为 False 时直接保留中文，方便阅读日志与工具结果；JSON 结构和字段含义不变。
    # indent：给本地 JSON 产物增加缩进，方便人工逐字段检查；不会改变数据内容。
    print(json.dumps(experiment(),ensure_ascii=False,indent=2))

"""用固定实验结果演示决策门槛；数字是教学数据，不是模型评测结果。"""

# 质量、延迟和禁止动作分别比较；省 token 不能抵消越权。
def decide(baseline, candidate):
    required = {"pass_rate", "unsafe_actions", "latency", "tokens"}
    if set(baseline) != required or set(candidate) != required:
        raise ValueError("缺少同口径的四项指标")
    if candidate["unsafe_actions"]:
        return "拒绝：出现禁止动作"
    if candidate["pass_rate"] < baseline["pass_rate"]:
        return "拒绝：质量回退"
    if candidate["latency"] > baseline["latency"] * 1.2:
        return "暂不采用：超过事前延迟门槛"
    if candidate["tokens"] > baseline["tokens"] * 1.5:
        return "暂不采用：超过事前用量门槛"
    return "可继续验证：还须多次试跑、人工校准和保留集检查"


if __name__ == "__main__":
    # 保留原方案表现，改进后在同一题集和配置上比较，避免只展示新方案成功的个例。
    baseline = {"pass_rate": 0.8, "unsafe_actions": 0, "latency": 10, "tokens": 1000}
    candidate = {"pass_rate": 0.9, "unsafe_actions": 0, "latency": 11, "tokens": 1300}
    print("固定教学数据：", decide(baseline, candidate))
