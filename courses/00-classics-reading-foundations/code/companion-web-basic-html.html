<h1>一起读《狂人日记》</h1><p>提出一个问题，回答下方可以查看用到的讲稿。</p>
<form id="form"><label for="question">你想了解什么？</label><textarea id="question" maxlength="800" required>这份讲稿怎样解释文言文序言和白话日记？</textarea><button id="submit">查资料并回答</button></form>
<p id="status" role="status"></p><article id="result" hidden><p id="answer"></p><div id="sources"></div></article>
<script>
document.getElementById('form').addEventListener('submit',async event=>{
 event.preventDefault();const button=document.getElementById('submit');button.disabled=true;
 document.getElementById('result').hidden=true;
 document.getElementById('status').textContent='正在查找资料并组织回答……';
 try{
  const response=await fetch('/api/ask',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({question:document.getElementById('question').value,book:'狂人日记'})});
  const result=await response.json();if(!response.ok)throw new Error(typeof result.detail==='string'?result.detail:'请求未通过检查');
  // 用 textContent 显示不可信模型输出；其中即使有 HTML，也不会被浏览器执行。
  document.getElementById('answer').textContent=result.answer;const list=document.getElementById('sources');list.replaceChildren();
  for(const source of result.sources||[]){const details=document.createElement('details'),summary=document.createElement('summary'),text=document.createElement('p');summary.textContent=source.file+' · '+source.parent;text.textContent=source.text;details.append(summary,text);list.append(details);}
  document.getElementById('result').hidden=false;document.getElementById('status').textContent='回答已生成，请对照来源判断。';
 }catch(error){document.getElementById('status').textContent=error.message;}
 finally{button.disabled=false;}
});
</script></html>
